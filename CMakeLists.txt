cmake_minimum_required(VERSION 3.20)
project(cabinet_status VERSION 0.2.2 LANGUAGES C)
if(NOT WIN32)
    message(FATAL_ERROR "Build the Windows DLL with a Windows compiler.")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(cabinet-status SHARED src/cabinet_windows.c src/cabinet_core.c)
target_link_libraries(cabinet-status PRIVATE ws2_32)
set_target_properties(cabinet-status PROPERTIES PREFIX "")
add_executable(core_tests tests/core_tests.c src/cabinet_core.c)
target_include_directories(core_tests PRIVATE src)
add_executable(runtime_tests tests/runtime_tests.c)
target_link_libraries(runtime_tests PRIVATE ws2_32)
add_library(late_fixture SHARED tests/late_fixture.c)
set_target_properties(late_fixture PROPERTIES PREFIX "")
add_dependencies(runtime_tests late_fixture)
if(MSVC)
    target_compile_definitions(cabinet-status PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_definitions(runtime_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(cabinet-status PRIVATE /W4)
else()
    target_compile_options(cabinet-status PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME core COMMAND core_tests)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(TEST_PORT 28201)
else()
    set(TEST_PORT 28202)
endif()
add_test(NAME runtime COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass
    -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/run-runtime.ps1"
    -DllPath "$<TARGET_FILE:cabinet-status>"
    -TestExe "$<TARGET_FILE:runtime_tests>"
    -TestDirectory "${CMAKE_CURRENT_BINARY_DIR}/runtime-test"
    -Port ${TEST_PORT})
set_tests_properties(runtime PROPERTIES TIMEOUT 75)
install(TARGETS cabinet-status RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES cabinet-status.ini start.bat run-amdaemon.bat DESTINATION .)
